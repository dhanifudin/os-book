%% \indexterm{X} writes an index entry that:
%%   - sorts under the lowercased form of X (so chapter uses of \term{driver}
%%     merge with glossary uses of \gitem{Driver} under a single entry),
%%   - displays X with its first character forced to uppercase, so the index
%%     reads "Driver" / "ACL" / "Kernel space" regardless of how the source
%%     wrote it.
\makeatletter
\newcommand{\indexterm}[1]{%
    \begingroup
        \lowercase{\edef\@@idxkey{#1}}%
        \@@idxcapfirst#1\@nil
        \protected@edef\@@idxentry
            {\noexpand\index{\@@idxkey @\@@idxdisp}}%
        \@@idxentry
    \endgroup
}
\def\@@idxcapfirst#1#2\@nil{%
    \uppercase{\def\@@idxfirst{#1}}%
    \edef\@@idxdisp{\@@idxfirst#2}%
}
\makeatother
\newcommand{\term}[1]{\textit{#1}\protect\indexterm{#1}}
\newcommand{\gitem}[2]{\item[\textit{#1}]\indexterm{#1} #2}
