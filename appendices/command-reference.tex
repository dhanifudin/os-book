\section{Perintah Sistem File}

\subsection{ls - List Directory Contents}
\begin{lstlisting}[language=bash]
ls -la          # List semua file dengan detail
ls -lh          # List dengan ukuran human-readable
ls -lt          # Sort berdasarkan waktu modifikasi
\end{lstlisting}

\subsection{cd - Change Directory}
\begin{lstlisting}[language=bash]
cd /path/to/dir # Pindah ke direktori
cd ..           # Pindah ke parent directory
cd ~            # Pindah ke home directory
cd -            # Pindah ke direktori sebelumnya
\end{lstlisting}

\subsection{pwd - Print Working Directory}
\begin{lstlisting}[language=bash]
pwd             # Tampilkan direktori saat ini
\end{lstlisting}
